Запуск: pip install aiogram → export BOT_TOKEN=... → python perekup_bot.py """ import asyncio import json import logging import os import random from aiogram import Bot, Dispatcher, F from aiogram.exceptions import TelegramBadRequest from aiogram.filters import Command, CommandStart from aiogram.types import CallbackQuery, InlineKeyboardButton, InlineKeyboardMarkup, Message TOKEN = os.getenv("BOT_TOKEN", "") SAVE_FILE = "perekup_save.json" # ---------- Настройки баланса ---------- START_MONEY = 5000 GOAL = 100_000 # цель: накопить столько MAX_DAYS = 30 # за столько дней ENERGY = 6 # действий в день RENT = 150 # аренда гаража в день BASE_CAP = 3 # слотов склада в начале DEFECT_CHANCE = 0.25 # шанс скрытого дефекта у лота ITEMS = { "📱 iPhone 11": 15000, "🎮 PlayStation 4": 12000, "🚲 Велосипед": 8000, "👟 Кроссовки Nike": 5000, "💻 Ноутбук": 25000, "🎧 AirPods": 6000, "🎸 Гитара": 10000, "🧥 Куртка": 4000, "⌚ Часы": 7000, "🖥 Монитор": 9000, "📷 Фотоаппарат": 14000, "🛴 Самокат": 6500, } players: dict[str, dict] = {} # ---------- Хранение ---------- def load(): global players if os.path.exists(SAVE_FILE): with open(SAVE_FILE, encoding="utf-8") as f: players = json.load(f) def save(): with open(SAVE_FILE, "w", encoding="utf-8") as f: json.dump(players, f, ensure_ascii=False) def fmt(n) -> str: return f"{int(n):,}".replace(",", " ") + " ₽" def r10(x) -> int: return max(10, int(round(x, -1))) # ---------- Логика игры ---------- def roll_prices(p): p["prices"] = {n: r10(b * random.uniform(0.85, 1.2)) for n, b in ITEMS.items()} p["event"] = None if random.random() < 0.35: name = random.choice(list(ITEMS)) if random.random() < 0.5: p["prices"][name] = r10(p["prices"][name] * 1.35) p["event"] = f"Хайп! {name} резко подорожал." else: p["prices"][name] = r10(p["prices"][name] * 0.7) p["event"] = f"Рынок завален: {name} подешевел." def new_player() -> dict: p = { "money": START_MONEY, "day": 1, "energy": ENERGY, "rep": 0, "cap": BASE_CAP, "inv": [], "offer": None, "prices": {}, "event": None, "over": False, "deals": 0, } roll_prices(p) return p def get_player(uid) -> dict: uid = str(uid) if uid not in players: players[uid] = new_player() return players[uid] def haggle_chance(p) -> float: return 0.45 + min(p["rep"], 20) * 0.01 def real_value(p, it) -> int: v = p["prices"][it["item"]] return int(v * (0.6 if it["defect"] else 1)) def net_worth(p) -> int: return p["money"] + sum(real_value(p, it) for it in p["inv"]) def upgrade_cost(p) -> int: return 2000 * (p["cap"] - 2) # ---------- Клавиатуры ---------- def kb(*rows) -> InlineKeyboardMarkup: return InlineKeyboardMarkup( inline_keyboard=[[InlineKeyboardButton(text=t, callback_data=d) for t, d in row] for row in rows] ) def menu_kb(p): return kb( [("🔍 Искать лот (1⚡)", "search")], [("📦 Склад / продать", "inv"), ("🏷 Цены рынка", "market")], [(f"🏗 Склад +2 ({fmt(upgrade_cost(p))})", "upgrade")], [("😴 Закончить день", "next")], ) # ---------- Тексты ---------- def header(p) -> str: return ( f"📅 День {p['day']}/{MAX_DAYS} ⚡ {p['energy']}/{ENERGY}\n" f"💵 {fmt(p['money'])} ⭐ Репутация: {p['rep']}\n" f"📦 Склад: {len(p['inv'])}/{p['cap']}\n" f"🎯 Цель: {fmt(GOAL)}" ) def menu_text(p, note="") -> str: ev = f"\n\n📰 {p['event']}" if p["event"] else "" return (f"{note}\n\n" if note else "") + header(p) + ev def offer_text(p) -> str: o = p["offer"] t = f"🛒 Продавец предлагает: {o['item']}\n💰 Цена: {fmt(o['ask'])}\n📊 Рынок: ~{fmt(p['prices'][o['item']])}\n" if o["checked"]: t += "⚠️ Проверка: скрытый дефект! Реальная цена ~60% от рынка.\n" if o["defect"] else "✅ Проверка: дефектов нет.\n" else: t += "🤔 Состояние по фото — вроде нормальное...\n" return t + f"\n💵 У вас: {fmt(p['money'])} ⚡ {p['energy']}" def sell_text(p) -> str: o = p["offer"] it = p["inv"][o["idx"]] t = ( f"🤝 Покупатель хочет: {it['item']}\n" f"💰 Предлагает: {fmt(o['bid'])}\n" f"📊 Рынок: ~{fmt(p['prices'][it['item']])}\n" f"🧾 Вы купили за: {fmt(it['cost'])}\n" ) if it["known"] and it["defect"]: t += "⚠️ У вещи дефект (вы знаете).\n" return t # ---------- Отправка ---------- async def show(event, text, markup): if isinstance(event, CallbackQuery): try: await event.message.edit_text(text, reply_markup=markup) except TelegramBadRequest: pass await event.answer() else: await event.answer(text, reply_markup=markup) async def show_menu(event, p, note=""): p["offer"] = None save() if p["over"]: await show(event, note, kb([("🔄 Новая игра", "restart")])) else: await show(event, menu_text(p, note), menu_kb(p)) def finish(p, text): p["over"] = True return text # ---------- Хендлеры ---------- dp = Dispatcher() @dp.message(CommandStart()) async def cmd_start(m: Message): p = get_player(m.from_user.id) intro = ( "🔥 ПЕРЕКУП 🔥\n\n" f"Начинаешь с {fmt(START_MONEY)}. Покупай вещи дешевле рынка, " f"продавай дороже и накопи {fmt(GOAL)} за {MAX_DAYS} дней.\n" f"Каждый день: {ENERGY} действий (⚡) и аренда {fmt(RENT)}.\n" "Следи за скрытыми дефектами и торгуйся — репутация повышает шансы!" ) await m.answer(intro) await show_menu(m, p) @dp.message(Command("new")) async def cmd_new(m: Message): players[str(m.from_user.id)] = new_player() await show_menu(m, players[str(m.from_user.id)], "Новая игра начата!") @dp.callback_query() async def on_cb(cb: CallbackQuery): p = get_player(cb.from_user.id) data = cb.data or "" act, _, arg = data.partition(":") if act == "restart": players[str(cb.from_user.id)] = p = new_player() return await show_menu(cb, p, "Новая игра!") if p["over"]: return await show_menu(cb, p, "Игра окончена. Нажми «Новая игра».") o = p["offer"] if act == "menu": return await show_menu(cb, p) if act == "market": lines = [] for n, base in ITEMS.items(): cur = p["prices"][n] arrow = "📈" if cur > base * 1.05 else "📉" if cur < base * 0.95 else "➖" lines.append(f"{arrow} {n}: {fmt(cur)}") return await show(cb, "🏷 Цены на рынке сегодня:\n\n" + "\n".join(lines), kb([("⬅️ Назад", "menu")])) if act == "search": if p["energy"] < 1: return await show_menu(cb, p, "😵 Нет сил. Закончи день.") if len(p["inv"]) >= p["cap"]: return await show_menu(cb, p, "📦 Склад полон! Продай что-нибудь или расширь склад.") p["energy"] -= 1 item = random.choice(list(ITEMS)) ask = r10(p["prices"][item] * random.uniform(0.55, 1.0)) p["offer"] = {"type": "buy", "item": item, "ask": ask, "defect": random.random() < DEFECT_CHANCE, "checked": False, "haggled": False} save() return await show(cb, offer_text(p), buy_kb(p)) if act in ("buy", "haggle_buy", "inspect", "skip") and (not o or o["type"] != "buy"): return await show_menu(cb, p, "Это предложение уже неактуально.") if act == "buy": if p["money"] < o["ask"]: return await show(cb, offer_text(p) + "\n❌ Не хватает денег!", buy_kb(p)) p["money"] -= o["ask"] p["inv"].append({"item": o["item"], "cost": o["ask"], "defect": o["defect"], "known": o["checked"]}) return await show_menu(cb, p, f"✅ Куплено: {o['item']} за {fmt(o['ask'])}") if act == "haggle_buy": o["haggled"] = True if random.random() < haggle_chance(p): o["ask"] = r10(o["ask"] * random.uniform(0.80, 0.92)) save() return await show(cb, "😎 Продавец уступил!\n\n" + offer_text(p), buy_kb(p)) if random.random() < 0.4: return await show_menu(cb, p, "😤 Продавец обиделся и ушёл.") save() return await show(cb, "🙅 Продавец не уступает.\n\n" + offer_text(p), buy_kb(p)) if act == "inspect": if p["energy"] < 1: return await show(cb, offer_text(p) + "\n😵 Нет сил на проверку.", buy_kb(p)) p["energy"] -= 1 o["checked"] = True save() return await show(cb, offer_text(p), buy_kb(p)) if act == "skip": return await show_menu(cb, p, "Пропустили лот.") if act == "inv": if not p["inv"]: return await show(cb, "📦 Склад пуст. Найди что-нибудь!", kb([("⬅️ Назад", "menu")])) rows = [[(f"{it['item']} (за {fmt(it['cost'])})", f"sell:{i}")] for i, it in enumerate(p["inv"])] rows.append([("⬅️ Назад", "menu")]) return await show(cb, "📦 Что продаём? (поиск покупателя = 1⚡)", kb(*rows)) if act == "sell": i = int(arg) if i >= len(p["inv"]): return await show_menu(cb, p, "Вещь не найдена.") if p["energy"] < 1: return await show_menu(cb, p, "😵 Нет сил искать покупателя.") p["energy"] -= 1 it = p["inv"][i] bid = r10(real_value(p, it) * random.uniform(0.75, 1.1) * (1 + min(p["rep"], 20) * 0.005)) p["offer"] = {"type": "sell", "idx": i, "bid": bid, "haggled": False} save() return await show(cb, sell_text(p), sell_kb(p)) if act in ("accept_sell", "haggle_sell", "decline_sell") and (not o or o["type"] != "sell"): return await show_menu(cb, p, "Покупатель уже ушёл.") if act == "accept_sell": it = p["inv"].pop(o["idx"]) p["money"] += o["bid"] profit = o["bid"] - it["cost"] p["deals"] += 1 p["rep"] += 1 if profit > 0 else 0 sign = "📈 Прибыль" if profit >= 0 else "📉 Убыток" note = f"💸 Продано: {it['item']} за {fmt(o['bid'])}\n{sign}: {fmt(abs(profit))}" if it["defect"] and not it["known"]: note += "\n(у вещи был скрытый дефект — покупатель сбил цену)" if p["money"] >= GOAL: return await show_menu(cb, p, finish(p, f"🏆 ПОБЕДА! У тебя {fmt(p['money'])} на день {p['day']}.\nСделок: {p['deals']}")) return await show_menu(cb, p, note) if act == "haggle_sell": o["haggled"] = True if random.random() < haggle_chance(p): o["bid"] = r10(o["bid"] * random.uniform(1.08, 1.2)) save() return await show(cb, "😏 Покупатель согласился на больше!\n\n" + sell_text(p), sell_kb(p)) if random.random() < 0.35: return await show_menu(cb, p, "🚶 Покупатель ушёл.") save() return await show(cb, "🙅 Покупатель стоит на своём.\n\n" + sell_text(p), sell_kb(p)) if act == "decline_sell": return await show_menu(cb, p, "Отказали покупателю.") if act == "upgrade": cost = upgrade_cost(p) if p["money"] < cost: return await show_menu(cb, p, f"Нужно {fmt(cost)} на расширение.") p["money"] -= cost p["cap"] += 2 return await show_menu(cb, p, "🏗 Склад расширен на 2 слота!") if act == "next": p["day"] += 1 p["energy"] = ENERGY p["money"] -= RENT if p["money"] < 0: return await show_menu(cb, p, finish(p, f"💀 Банкрот! Капитал: {fmt(nw)} из {fmt(GOAL)}. Попробуй ещё раз!" return await show_menu(cb, p, finish(p, txt)) roll_prices(p) return await show_menu(cb, p, f"🌅 Новый день. Аренда: -{fmt(RENT)}") await cb.answer() def buy_kb(p): o = p["offer"] row2 = [] if not o["haggled"]: row2.append(("🗣 Торговаться", "haggle_buy")) if not o["checked"]: row2.append(("🔎 Проверить (1⚡)", "inspect")) rows = [[("✅ Купить", "buy"), ("❌ Пропустить", "skip")]] if row2: rows.append(row2) return kb(*rows) def sell_kb(p): o = p["offer"] rows = [[("✅ Продать", "accept_sell"), ("❌ Отказать", "decline_sell")]] if not o["haggled"]: rows.append([("🗣 Набить цену", "haggle_sell")]) return kb(*rows) async def main(): if not TOKEN: raise SystemExit("Задай токен: export BOT_TOKEN=123:ABC...") logging.basicConfig(level=logging.INFO) load() bot = Bot(TOKEN) await dp.start_polling(bot) if __name__ == "__main__": asyncio.run(main())
